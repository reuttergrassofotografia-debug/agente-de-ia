# API Webhook Server (apps/api) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Fastify webhook server that receives Evolution API events, validates them, persists messages to Supabase, and enqueues jobs in BullMQ.

**Architecture:** Single Fastify app in `apps/api`. Validates the `apikey` header against the instance's `webhook_secret` in Supabase. Parses and validates the payload with Zod. Only handles text messages (`messages.upsert` with `fromMe: false`). Idempotent via `evolution_message_id`. Enqueues jobs via `@agente/queue`.

**Tech Stack:** Fastify 4.x, Zod 3.x, @agente/db, @agente/queue, @agente/evolution, tsx (dev), Vitest 1.x

## Global Constraints
- Node.js ≥ 20.0.0
- TypeScript `strict: true`
- `"type": "module"` in package.json
- Named exports only
- Import paths use `.js` extension
- All env vars loaded via `process.env` with validation at startup
- Webhook endpoint: `POST /webhook`
- Returns `200` for all valid events (even skipped ones) — Evolution API retries on non-2xx
- Returns `401` for invalid apikey, `400` for malformed payload, `500` for unexpected errors

## Prerequisites
Plan 1 (monorepo + packages) must be completed. Run `npm install` from root before starting this plan.

export function loadEnv() {
  const required = [
    'SUPABASE_URL',
    'SUPABASE_SERVICE_ROLE_KEY',
    'REDIS_URL',
    'EVOLUTION_API_URL',
    'EVOLUTION_API_KEY',
  ] as const

  const missing = required.filter((k) => !process.env[k])
  if (missing.length > 0) {
    throw new Error(`Missing required environment variables: ${missing.join(', ')}`)
  }

  return {
    SUPABASE_URL: process.env['SUPABASE_URL']!,
    SUPABASE_SERVICE_ROLE_KEY: process.env['SUPABASE_SERVICE_ROLE_KEY']!,
    REDIS_URL: process.env['REDIS_URL']!,
    EVOLUTION_API_URL: process.env['EVOLUTION_API_URL']!,
    EVOLUTION_API_KEY: process.env['EVOLUTION_API_KEY']!,
    PORT: Number(process.env['PORT'] ?? 3000),
  }
}

export type Env = ReturnType<typeof loadEnv>

- [ ] **Step 4: Create `apps/api/src/routes/webhook.ts`**

```typescript
import type { FastifyInstance } from 'fastify'
import type { SupabaseClient } from '@supabase/supabase-js'
import type { Queue } from 'bullmq'
import {
  getInstanceByName,
  getAgentByInstanceId,
  getOrCreateContact,
  getOrCreateConversation,
  createMessage,
  type Database,
} from '@agente/db'
import { enqueueMessage, type MessageJob } from '@agente/queue'
import { extractMessageText } from '@agente/evolution'
import { WebhookPayloadSchema } from '../schemas/webhook.js'

interface WebhookDeps {
  db: SupabaseClient<Database>
  queue: Queue<MessageJob>
}

export function registerWebhookRoute(app: FastifyInstance, { db, queue }: WebhookDeps): void {
  app.post('/webhook', async (request, reply) => {
    // Parse and validate payload
    const parseResult = WebhookPayloadSchema.safeParse(request.body)
    if (!parseResult.success) {
      return reply.status(400).send({ error: 'Invalid payload', details: parseResult.error.format() })
    }
    const payload = parseResult.data

    // Only handle incoming text messages
    if (payload.data.key.fromMe) return reply.status(200).send({ ok: true, skipped: 'fromMe' })

    // Validate apikey against instance webhook_secret
    const apikey = (request.headers['apikey'] as string | undefined)
    const instance = await getInstanceByName(db, payload.instance)
    if (!apikey || !instance || apikey !== instance.webhook_secret) {
      return reply.status(401).send({ error: 'Unauthorized' })
    }

    // Extract text — skip if not a text message
    const text = extractMessageText(payload)
    if (!text) return reply.status(200).send({ ok: true, skipped: 'non-text' })

    // Only process if active agent exists
    const agent = await getAgentByInstanceId(db, instance.id)
    if (!agent) return reply.status(200).send({ ok: true, skipped: 'no-agent' })

    // Derive phone from remoteJid (strip @s.whatsapp.net)
    const phone = payload.data.key.remoteJid.split('@')[0] ?? payload.data.key.remoteJid

    // Persist contact, conversation, message
    const contact = await getOrCreateContact(db, instance.id, phone, payload.data.pushName)
    const conversation = await getOrCreateConversation(db, contact.id, instance.id, agent.id)
    const message = await createMessage(db, {
      conversation_id: conversation.id,
      role: 'user',
      content: text,
      evolution_message_id: payload.data.key.id,
    })

    // Enqueue job
    await enqueueMessage(queue, {
      instanceId: instance.id,
      contactId: contact.id,
      messageId: message.id,
      conversationId: conversation.id,
      evolutionInstanceName: payload.instance,
      contactPhone: phone,
    })

    return reply.status(200).send({ ok: true, jobEnqueued: true })
  })
}
```
